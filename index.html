[primary_image] : []), ...moreImages ].slice(0, 30); return { offer_id, name, category_id: currentC2, description: commonDesc, price: price, currency_code: currency, barcode: barcode || undefined, depth, width, height, dimension_unit: 'cm', weight, weight_unit: 'kg', primary_image: primary_image || undefined, images: images.map(u => ({ file_name: u })), attributes: attrsCommon }; } $tbody.addEventListener('click', async (e) => { const btn = e.target; const row = btn.closest('tr[data-row]'); if (!row) return; // （已有）删除行 if (btn.classList.contains('del-row')) { row.remove(); return; } // （已有）上传到 sm.ms if (btn.classList.contains('upload-smms')) { // ... 你现有的 sm.ms 上传逻辑保持不变 ... return; } // 🆕 立即发布（只发这一行） if (btn.classList.contains('publish-one')) { try { if (!currentC2 || !currentType) { alert('请先选择类目（到第三级类型）并等待属性加载完成。'); return; } const attrsCommon = collectAttributesFromUI(); const commonDesc = $commonDesc.value || ''; const item = buildItemFromRow(row, attrsCommon, commonDesc); // 简单校验 if (!item.offer_id || !item.name || !item.price || !item.depth || !item.width || !item.height || !item.weight) { alert('请检查：offer_id / 标题 / 价格 / 长宽高 / 重量 必填且非 0'); return; } btn.disabled = true; const oldText = btn.textContent; btn.textContent = '发布中...'; const resp = await ozon('/v3/product/import', { items: [item] }); $batchResult.textContent = JSON.stringify(resp, null, 2); btn.textContent = '已提交'; setTimeout(() => { btn.textContent = oldText; btn.disabled = false; }, 1500); } catch (err) { btn.disabled = false; alert('发布失败：' + (err.message || err)); } } });
